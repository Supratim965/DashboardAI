2    import pandas as pd
3    from openpyxl import Workbook
4    from openpyxl.chart import BarChart, LineChart, PieChart, Reference
5    from langchain_community.llms import HuggingFacePipeline
6    from langchain_core.prompts import PromptTemplate
7    from langchain.chains.llm import LLMChain
8    from transformers import pipeline, AutoTokenizer, AutoModelForCausalLM
9    import torch
10   
11   # Load free LLM (e.g., Mistral-7B - runs on CPU/GPU)
12   model_id = "TinyLlama/TinyLlama-1.1B-Chat-v1.0"
13   tokenizer = AutoTokenizer.from_pretrained(model_id)
14   model = AutoModelForCausalLM.from_pretrained(model_id, torch_dtype=torch.float16, device_map="auto")
15   pipe = pipeline("text-generation", model=model, tokenizer=tokenizer, max_new_tokens=200)
16   llm = HuggingFacePipeline(pipeline=pipe)
17   
18   # Sample data (replace with your CSV/Excel)
19   df = pd.DataFrame({
20       'Month': ['Jan', 'Feb', 'Mar', 'Apr'],
21       'Sales': [100, 150, 120, 180],
22       'Region': ['North', 'South', 'North', 'South']
23   })
24   
25   # AI Agent: Analyze and suggest dashboard elements
26   prompt_template = PromptTemplate(
27       input_variables=["data_summary"],
28       template="""Analyze this data summary: {data_summary} 
29   Suggest 2-3 dashboard elements (charts/tables) with exact instructions like: 
30   1. Bar chart: Sales by Month (column A vs B) 
31   2. Pie chart: Sales by Region (group C) 
32   3. Summary table: Total sales by region 
33   Respond ONLY with numbered instructions."""
34   )
35   
36   chain = LLMChain(llm=llm, prompt=prompt_template)
37   data_summary = df.describe().to_string() + "Columns: " + str(df.columns.tolist())
38   instructions = chain.run(data_summary)
39   print("AI Suggestions:", instructions)
40   # e.g., "1. Bar chart: Sales by Month..."
41   
42   # Create Excel dashboard (parse AI instructions or use defaults)
43   wb = Workbook()
44   ws_data = wb.active
45   ws_data.title = 'Data'
46   for r, row in enumerate(df.itertuples(index=False), 1):
47       for c, value in enumerate(row, 1):
48           ws_data.cell(row=r, column=c, value=value)
49   
50   ws_dash = wb.create_sheet('Dashboard')
51   ws_dash['A1'] = 'AI-Generated Sales Dashboard'
52   
53   # AI-driven charts (parse instructions or auto-generate based on trends)
54   # Example: Detect upward trend → Line chart
55   if df['Sales'].is_monotonic_increasing:
56       chart = LineChart()
57   else:
58       chart = BarChart()
59   data_ref = Reference(ws_data, min_col=2, min_row=1, max_row=len(df) + 1, max_col=2)
60   cats = Reference(ws_data, min_col=1, min_row=2, max_row=len(df) + 1)
61   chart.add_data(data_ref)
62   chart.set_categories(cats)
63   chart.title = 'AI Selected: Sales Trend'
64   ws_dash.add_chart(chart, 'E5')
65   
66   # Summary table (Pivot-like)
67   pivot = df.pivot_table(values='Sales', index='Region', aggfunc='sum')
68   for r, (region, total) in enumerate(pivot.iterrows(), 6):
69       ws_dash[f'A{r}'] = region
70       ws_dash[f'B{r}'] = total
71   
72   wb.save('ai_dashboard.xlsx')
73   print("AI dashboard saved! Opens in Excel 2007")
