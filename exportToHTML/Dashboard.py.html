8    # Sample data (replace with pd.read_csv('yourfile.csv') or pd.read_excel)
9    df = pd.DataFrame({

15   # Create workbook and data sheet
16   wb = Workbook()

18   ws_data.title = 'Data'
19   for r, row in enumerate(df.itertuples(index=False), 1):
20       for c, value in enumerate(row, 1):
21           ws_data.cell(row=r, column=c, value=value)
22   
23   # Dashboard sheet
24   ws_dash = wb.create_sheet('Dashboard')
25   
26   # Title
27   ws_dash.merge_cells('A1:D4')
28   title_cell = ws_dash['A1']
29   title_cell.value = 'Sales Dashboard'
30   title_cell.font = Font(bold=True, size=20)
31   title_cell.alignment = Alignment(horizontal='center')
32   title_cell.fill = PatternFill('solid', fgColor='2591DB')
33   
34   # Copy summary table (e.g., totals by region)
35   summary = df.groupby('Region')['Sales'].sum()
36   for c, (region, total) in enumerate(summary.items(), 1):
37       ws_dash[f'A5'] = 'Region'
38       ws_dash[f'B5'] = region
39       ws_dash[f'C5'] = total
40   
41   # Bar chart for sales by month
42   chart1 = BarChart()
43   data = Reference(ws_data, min_col=2, min_row=1, max_row=5, max_col=2)
44   cats = Reference(ws_data, min_col=1, min_row=2, max_row=5)
45   chart1.add_data(data)
46   chart1.set_categories(cats)

48   ws_dash.add_chart(chart1, 'E5')
